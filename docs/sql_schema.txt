DROP DATABASE IF EXISTS restaurant_guide;
CREATE DATABASE restaurant_guide;
USE restaurant_guide;

CREATE TABLE app_user(
  user_id INT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(255) NOT NULL,
  email    VARCHAR(255) NOT NULL UNIQUE,
  user_password VARCHAR(255) NOT NULL,
    role     ENUM('user', 'owner', 'admin') NOT NULL DEFAULT 'user',
  is_blocked BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE area (
  area_id   INT AUTO_INCREMENT PRIMARY KEY,
  area_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE place (
  place_id    INT AUTO_INCREMENT PRIMARY KEY,
  place_name  VARCHAR(255) NOT NULL,
  phone_number VARCHAR(255),
  latitude    DECIMAL(9,6) NOT NULL,
  longitude   DECIMAL(9,6) NOT NULL,
  rating      DECIMAL(2,1),
  price_range VARCHAR(255),
  status      ENUM('draft', 'published', 'hidden') NOT NULL DEFAULT 'draft',
  area_id     INT,
  owner_id    INT,
  FOREIGN KEY (area_id) REFERENCES area(area_id) ON DELETE SET NULL,
  FOREIGN KEY (owner_id) REFERENCES app_user(user_id) ON DELETE SET NULL
);

CREATE TABLE category(
  category_id INT AUTO_INCREMENT PRIMARY KEY,
  category_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE place_category (
  place_id    INT NOT NULL,
  category_id INT NOT NULL,
  PRIMARY KEY (place_id, category_id),
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE,
  FOREIGN KEY (category_id) REFERENCES category(category_id) ON DELETE CASCADE
);

CREATE TABLE place_hours (
  PRIMARY KEY (place_id, day_of_week),
  place_id    INT NOT NULL,
  day_of_week TINYINT,      -- 0 = Sunday  6 = Saturday
  open_time   TIME,
  close_time  TIME,
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE
);

CREATE TABLE meal(
  meal_id INT AUTO_INCREMENT PRIMARY KEY,
  meal_name VARCHAR(255),
  place_id INT,
  price DECIMAL(5,2),
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE
);

CREATE TABLE meal_category(
  meal_id     INT NOT NULL,
  category_id INT NOT NULL,
  PRIMARY KEY (meal_id, category_id),
  FOREIGN KEY (meal_id) REFERENCES meal(meal_id) ON DELETE CASCADE,
  FOREIGN KEY (category_id) REFERENCES category(category_id) ON DELETE CASCADE
);

CREATE TABLE attribute(
  attribute_id   INT AUTO_INCREMENT PRIMARY KEY,
  attribute_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE place_attribute(
  PRIMARY KEY (place_id, attribute_id),
  place_id     INT NOT NULL,
  attribute_id INT NOT NULL,
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE,
  FOREIGN KEY (attribute_id) REFERENCES attribute(attribute_id) ON DELETE CASCADE
);

CREATE TABLE review (
  review_id   INT AUTO_INCREMENT PRIMARY KEY,
  user_id     INT NOT NULL,
  place_id    INT NOT NULL,
  rating      TINYINT NOT NULL,
  comment     TEXT,
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE (user_id, place_id),
  FOREIGN KEY (user_id) REFERENCES app_user(user_id) ON DELETE CASCADE,
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE
);

CREATE TABLE favorite (
  user_id    INT NOT NULL,
  place_id   INT NOT NULL,
  PRIMARY KEY (user_id, place_id),
  FOREIGN KEY (user_id) REFERENCES app_user(user_id) ON DELETE CASCADE,
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE
);

CREATE TABLE occasion (
  occasion_id   INT AUTO_INCREMENT PRIMARY KEY,
  occasion_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE place_occasion (
  place_id    INT NOT NULL,
  occasion_id INT NOT NULL,
  PRIMARY KEY (place_id, occasion_id),
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE,
  FOREIGN KEY (occasion_id) REFERENCES occasion(occasion_id) ON DELETE CASCADE
);

CREATE TABLE photo (
  photo_id   INT AUTO_INCREMENT PRIMARY KEY,
  place_id   INT NOT NULL,
  file_name  VARCHAR(255) NOT NULL,
  is_cover   BOOLEAN NOT NULL DEFAULT FALSE,
  sort_order INT NOT NULL DEFAULT 0,
  FOREIGN KEY (place_id) REFERENCES place(place_id) ON DELETE CASCADE
);

CREATE TABLE review_report (
  report_id  INT AUTO_INCREMENT PRIMARY KEY,
  review_id  INT NOT NULL,
  user_id    INT NOT NULL,
  reason     VARCHAR(255),
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE (review_id, user_id),
  FOREIGN KEY (review_id) REFERENCES review(review_id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES app_user(user_id) ON DELETE CASCADE
);
